% \cvrow{<date>}{<content>} -- date in a narrow left column, content in a wide right column
\newcommand{\cvrow}[2]{%
  \needspace{2\baselineskip}%
  \noindent\setlength\tabcolsep{0pt}%
  \cvhlbox{\begin{tabular*}{\linewidth}{@{\extracolsep{\fill}} L{2.6cm} L{\dimexpr\linewidth-2.6cm-1.4cm\relax}}
  {\color{black} #1} & #2 \\
  \end{tabular*}}\par
  \vspace{4pt}%
}

% \cvrowtight{<date>}{<content>} -- same layout as \cvrow, but no space after: for dense,
% single-line list items (Honors, Service, Memberships, Teaching)
\newcommand{\cvrowtight}[2]{%
  \needspace{1\baselineskip}%
  \noindent\setlength\tabcolsep{0pt}%
  \cvhlbox{\begin{tabular*}{\linewidth}{@{\extracolsep{\fill}} L{2.6cm} L{\dimexpr\linewidth-2.6cm-1.4cm\relax}}
  {\color{black} #1} & #2 \\
  \end{tabular*}}\par
  \vspace{-2pt}%
}

% \cveduc{<degree>}{<institution>}{<location>}{<date>} -- institution/degree on the left,
% location/date right-aligned on the right, matching the Education section layout
\newcommand{\cveduc}[4]{%
  \needspace{2\baselineskip}%
  \noindent\setlength\tabcolsep{0pt}%
  \cvhlbox{\begin{tabular*}{\linewidth}{@{\extracolsep{\fill}} L{\dimexpr\linewidth-4.6cm\relax} R{4.6cm}}
  \textbf{#2} & {\color{black} #3} \\
  #1 & {\color{black} #4} \\
  \end{tabular*}}\par
  \vspace{7.2pt}%
}
